# 3. 공통 함수: 정책 값을 안전하게 가져오는 도구
function Get-PolicyValue ($Pattern) {
    # 1. 파일 경로 확인
    $File = $script:SecPolFile
    if (-not (Test-Path $File)) { return $null }

    # Select-String으로 파일 직접 검색 
    $Match = Select-String -Path $File -Pattern $Pattern | Select-Object -First 1
    
    if ($Match) {
        try {
            # "Key = Value" 형태에서 등호(=) 뒤의 숫자만 추출
            return [int]$Match.Line.Split('=')[1].Trim()
        } catch {
            return $null
        }
    }
    return $null
}

"--------------------------------------------------------`n" | Out-File $ResultFile -Append -Encoding utf8

# 결과 출력 및 객체 저장 함수
function print_result($CODE, $ITEM_NAME, $RES, $DET){
    switch($RES){
        "양호" { $script:GoodCnt++ }
        "취약" { $script:VulnCnt++; $script:VulnUnit += $CODE }
        "체크필요" { $script:CheckCnt++; $script:CheckUnit += $CODE }
    }
    $resultObj = [PSCustomObject]@{ 번호 = $CODE; 항목명 = $ITEM_NAME; 결과 = $RES; 상세내용 = $DET }
    $script:ResultsList += $resultObj

    "[$CODE] $ITEM_NAME" | Out-File $ResultFile -Append -Encoding utf8
    "결과 : $RES" | Out-File $ResultFile -Append -Encoding utf8
    "상세 : $DET" | Out-File $ResultFile -Append -Encoding utf8
    "----------------------------------------------`n" | Out-File $ResultFile -Append -Encoding utf8
    Write-Host "[$CODE] 점검 완료 -> $RES"
}

    }

}

# --------------------------------------------------------------------------------
# W-09. 비밀번호 관리정책 설정
# --------------------------------------------------------------------------------
function W_09 {
    $CODE = "W-09"; $script:NAME[$CODE] = "비밀번호 관리정책 설정"
    
    # 1단계에서 수정한 함수로 값을 읽어옵니다.
    $Comp   = Get-PolicyValue "PasswordComplexity"
    $Len    = Get-PolicyValue "MinimumPasswordLength"
    $MaxAge = Get-PolicyValue "MaximumPasswordAge"
    $MinAge = Get-PolicyValue "MinimumPasswordAge"

    # [중요] 값을 못 읽어왔으면(Null) 0으로 설정하여 '취약'으로 유도
    if ($null -eq $Comp)   { $Comp = 0 }   
    if ($null -eq $Len)    { $Len = 0 }    
    if ($null -eq $MaxAge) { $MaxAge = 0 } 
    if ($null -eq $MinAge) { $MinAge = 0 } 

    $Info = "복잡성:$Comp, 길이:$Len, 최대기간:$MaxAge, 최소기간:$MinAge"

    # [판단 기준]
    # 1. 복잡성 == 1
    # 2. 길이 >= 8
    # 3. 최대 기간 <= 90 (0은 만료 없음이므로 취약)
    # 4. 최소 기간 >= 1
    if ($Comp -eq 1 -and $Len -ge 8 -and ($MaxAge -le 90 -and $MaxAge -gt 0) -and $MinAge -ge 1) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "모든 정책 만족 ($Info)"
    } else {
        $script:RESULT[$CODE] = "취약"
        $Reason = @()
        if ($Comp -ne 1) { $Reason += "복잡성 미사용" }
        if ($Len -lt 8) { $Reason += "길이 미달" }
        if ($MaxAge -gt 90 -or $MaxAge -eq 0) { $Reason += "교체주기($MaxAge) 미흡" }
        if ($MinAge -lt 1) { $Reason += "최소사용기간 미흡" }
        
        $script:DETAIL[$CODE] = "정책 미흡: $($Reason -join ', ') ($Info)"
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-11. 로컬 로그온 허용
# --------------------------------------------------------------------------------
function W_11 {
    $CODE = "W-11"; $script:NAME[$CODE] = "로컬 로그온 허용"
    
    # 1. 파일에서 'SeInteractiveLogonRight' 줄을 찾습니다.
    # $script:SecPolContent가 없거나 패턴을 못 찾으면 $LogonRight는 null이 됩니다.
    if ($script:SecPolContent) {
        $Match = $script:SecPolContent | Select-String "SeInteractiveLogonRight"
        if ($Match) {
            $LogonRight = $Match.ToString()
        } else {
            $LogonRight = $null
        }
    } else {
        $LogonRight = $null
    }

    # 값을 읽지 못했다면 '체크필요'로 분류
    if ($null -eq $LogonRight) {
        $script:RESULT[$CODE] = "체크필요"
        $script:DETAIL[$CODE] = "정책 값을 읽을 수 없습니다. (정책 파일 오류 또는 설정 없음)"
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }
    
    # 2. 취약 그룹(Everyone, Guests) 확인
    if (($LogonRight -match "S-1-1-0" -or $LogonRight -match "Everyone" -or $LogonRight -match "Guests") -or ($null -eq $LogonRight)) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "Everyone 또는 Guests 그룹에 로컬 로그온 권한이 있습니다."
    } else {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "로컬 로그온 권한이 적절하게 제한되어 있습니다."
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-16. 공유 권한 및 사용자 그룹 설정 (Everyone 권한 정밀 점검)

    $Shares = Get-SmbShare | Where-Object { $_.Name -notlike "*$" }

        foreach ($Share in $Shares) {
            # 2. 각 공유 폴더의 접근 권한 목록(ACL) 조회

            }

            }
        }
    }

    if ($VulnShares.Count -gt 0) {
        $script:RESULT[$CODE] = "취약"

# W-20. NetBIOS 바인딩 서비스 구동 점검
# --------------------------------------------------------------------------------
function W_20 {
    $CODE = "W-20"; $script:NAME[$CODE] = "NetBIOS 바인딩 서비스 구동 점검"
    
    # [점검 기준] 네트워크 어댑터 속성 > WINS > "NetBIOS over TCP/IP 사용 안 함" 선택 여부
    # Win32_NetworkAdapterConfiguration 클래스의 TcpipNetbiosOptions 속성 확인
    # 0 : DHCP 서버 설정 사용 (기본값) -> '취약'
    # 1 : NetBIOS 사용 (활성화) -> '취약'
    # 2 : NetBIOS 사용 안 함 (비활성화) -> '양호'

    $Adapters = Get-CimInstance Win32_NetworkAdapterConfiguration | Where-Object { $_.IPEnabled -eq $true }
    $VulnAdapters = @()

    foreach ($Adapter in $Adapters) {
        # 설정값이 2(사용 안 함)가 아닌 경우 모두 취약으로 판단
        if ($Adapter.TcpipNetbiosOptions -ne 2) {
            # 0인 경우(DHCP)와 1인 경우(사용)를 구분하여 상세 메시지 생성
            $StatusStr = if ($Adapter.TcpipNetbiosOptions -eq 0) { "DHCP사용(기본값)" } else { "사용함" }
            $VulnAdapters += "$($Adapter.Description) [$StatusStr]"
        }
    }

    if ($VulnAdapters.Count -gt 0) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "NetBIOS가 차단되지 않은 어댑터가 있습니다: " + ($VulnAdapters -join ", ")
    } else {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "모든 네트워크 어댑터에서 NetBIOS가 '사용 안 함'으로 설정되어 있습니다."
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}


# --------------------------------------------------------------------------------
# W-21. 암호화되지 않는 FTP 서비스 비활성화 (3rd Party 및 IIS 상황 통합 고려)
# --------------------------------------------------------------------------------
function W_21 {
    $CODE = "W-21"; $script:NAME[$CODE] = "암호화되지 않는 FTP 서비스 비활성화"
    
    # 1. Windows 기본 FTP 서비스(ftpsvc) 상태 확인
    $FTP = Get-Service "ftpsvc" -ErrorAction SilentlyContinue

    if ($null -eq $FTP -or $FTP.Status -ne "Running") {
        # 서비스가 꺼져 있으면'양호'

        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "Windows 기본 FTP 서비스가 중지되어 있습니다. (타사 FTP 사용 시에도 이 상태가 권장됨)"
    } else {
        # 2. 서비스가 켜져 있음 -> IIS 설정을 통해 SSL 강제 여부 확인
        # (ftpsvc는 IIS의 하위 구성요소로 관리됩니다)
        if (Get-Module -ListAvailable -Name WebAdministration) {
            try {
                Import-Module WebAdministration -ErrorAction SilentlyContinue
                
                # FTP 바인딩이 있는 모든 사이트의 SSL 플래그 검사
                $Sites = Get-ChildItem -Path IIS:\Sites
                $InsecureSites = @()

                foreach ($Site in $Sites) {
                    if ($Site.bindings.Collection.protocol -contains "ftp") {
                        # sslFlags: 0(None), 1(Allow), 2(Require), 3(Require 128)
                        $Val = Get-WebConfigurationProperty -Filter /system.applicationHost/sites/site[@name="$($Site.name)"]/ftpServer/security/ssl -Name sslFlags -PSPath IIS:\
                        
                        # SSL이 '필요(2)' 이상이 아니면 취약 목록에 추가
                        if ($null -eq $Val.Value -or $Val.Value -lt 2) {
                            $InsecureSites += $Site.name
                        }
                    }
                }

                if ($InsecureSites.Count -gt 0) {
                    $script:RESULT[$CODE] = "취약"
                    $script:DETAIL[$CODE] = "FTP 서비스가 구동 중이며, SSL 암호화가 강제되지 않은 사이트가 있습니다: $($InsecureSites -join ', ')"
                } else {
                    $script:RESULT[$CODE] = "양호"
                    $script:DETAIL[$CODE] = "FTP 서비스가 구동 중이나, 모든 사이트에 SSL 암호화가 강제 설정되어 있습니다."
                }

            } catch {
                # IIS 모듈 로드 실패 등 예외 발생 시
                $script:RESULT[$CODE] = "체크필요"
                $script:DETAIL[$CODE] = "FTP 서비스가 실행 중입니다. (IIS 설정 조회 실패로 SSL 적용 여부 수동 확인 필요)"
            }
        } else {
            # WebAdministration 모듈이 없는 경우 (매우 드문 케이스)
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "FTP 서비스가 실행 중입니다. (SSL 강제 설정 여부를 확인할 수 없어 취약으로 간주)"
        }
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# --------------------------------------------------------------------------------
# W-24. FTP 접근 제어 설정 (IP 제한 여부 확인)
# --------------------------------------------------------------------------------
function W_24 {
    $CODE = "W-24"; $script:NAME[$CODE] = "FTP 접근 제어 설정"
    
    # 1. FTP 서비스 실행 여부 확인
    $FTP = Get-Service "ftpsvc" -ErrorAction SilentlyContinue

    if ($null -eq $FTP -or $FTP.Status -ne "Running") {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "FTP 서비스가 실행 중이지 않아 점검이 필요 없습니다."
    } else {
        # 2. 서비스 실행 중 -> IIS 설정을 통해 IP 제한 설정 확인
        if (Get-Module -ListAvailable -Name WebAdministration) {
            try {
                Import-Module WebAdministration -ErrorAction SilentlyContinue
                
                $Sites = Get-ChildItem -Path IIS:\Sites
                $VulnSites = @()

                foreach ($Site in $Sites) {
                    if ($Site.bindings.Collection.protocol -contains "ftp") {
                        # 'FTP IPv4 주소 및 도메인 제한' 설정 조회 (ipSecurity)
                        # allowUnlisted: True(기본값, 모두 허용), False(모두 거부)
                        $IpSec = Get-WebConfigurationProperty -Filter /system.ftpServer/security/ipSecurity -Name allowUnlisted -PSPath IIS:\ -Location $Site.name
                        
                        # [점검 기준] "지정되지 않은 클라이언트 거부(allowUnlisted=False)"가 설정되어야 양호
                        # 값이 True이면 '모두 허용' 상태이므로 취약 (특정 IP만 거부하는 블랙리스트 방식일 수도 있으나, 화이트리스트 권장, 해당 내용은 체크 필요)
                        
                        if ($IpSec.Value -eq $true) {
                            $VulnSites += "$($Site.name)(모두허용)"
                        }
                    }
                }

                if ($VulnSites.Count -gt 0) {
                    $script:RESULT[$CODE] = "취약"
                    $script:DETAIL[$CODE] = "FTP 접근 제어(IP 제한)가 설정되지 않은 사이트가 있습니다: $($VulnSites -join ', ')"
                } else {
                    $script:RESULT[$CODE] = "양호"
                    $script:DETAIL[$CODE] = "모든 FTP 사이트에 대해 접근 제어(지정된 IP만 허용)가 설정되어 있습니다."
                }

            } catch {
                $script:RESULT[$CODE] = "체크필요"
                $script:DETAIL[$CODE] = "FTP 실행 중. IIS 설정 조회 실패로 IP 제한 여부를 확인할 수 없습니다."
            }
        } else {
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "FTP 실행 중. (접근 제어 설정 여부 스크립트 확인 불가)"
        }

# --------------------------------------------------------------------------------
# W-25. DNS Zone Transfer 설정 (영역 전송 제한 확인)
# --------------------------------------------------------------------------------
function W_25 {
    $CODE = "W-25"; $script:NAME[$CODE] = "DNS Zone Transfer 설정"
    
    # 1. DNS 서비스(DNS Server) 실행 여부 확인
    $DnsSvc = Get-Service "DNS" -ErrorAction SilentlyContinue

    if ($null -eq $DnsSvc -or $DnsSvc.Status -ne "Running") {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "DNS 서비스를 사용하지 않습니다. (서비스 중지됨)"
    } else {
        # 2. 서비스 실행 중 -> 설정 확인 필요
        # (A) PowerShell DNS 모듈 사용 시도 (가장 정확함)
        if (Get-Command Get-DnsServerZone -ErrorAction SilentlyContinue) {
            try {
                $Zones = Get-DnsServerZone
                $VulnZones = @()

                foreach ($Zone in $Zones) {
                    # ZoneTransfer 속성 확인
                    # "Any" 또는 "Unknown"(설정꼬임)인 경우 취약으로 간주
                    # 안전한 값: "None", "NamedServers", "SpecificIp"
                    if ($Zone.ZoneTransfer -eq "Any") {
                        $VulnZones += "$($Zone.ZoneName)(모두허용)"
                    }
                }

                if ($VulnZones.Count -gt 0) {
                    $script:RESULT[$CODE] = "취약"
                    $script:DETAIL[$CODE] = "Zone Transfer가 '모든 서버'로 허용된 영역이 있습니다: $($VulnZones -join ', ')"
                } else {
                    $script:RESULT[$CODE] = "양호"
                    $script:DETAIL[$CODE] = "모든 DNS 영역의 Zone Transfer가 제한(차단 또는 특정서버 지정)되어 있습니다."
                }
            } catch {
                # 모듈 에러 시 레지스트리 검사로 넘어감
                $UseReg = $true
            }
        } else {
            # 모듈 없으면 레지스트리 검사 사용
            $UseReg = $true
        }

        # (B) 레지스트리 검사 (모듈 없을 때 Fallback)
        if ($UseReg) {
            $RegPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\DNS Server\Zones"
            if (Test-Path $RegPath) {
                $Zones = Get-ChildItem $RegPath
                $VulnZonesReg = @()
                
                foreach ($Z in $Zones) {
                    $Secure = Get-ItemProperty -Path $Z.PSPath -Name "SecureSecondaries" -ErrorAction SilentlyContinue
                    
                    # SecureSecondaries 값 의미
                    # 0: 모든 서버에 전송 허용 (취약) 
                    # 1: Name Servers 탭의 서버만 허용 (양호)
                    # 2: 지정한 IP만 허용 (양호)
                    # 3: 전송 안 함 (양호)
                    
                    if ($Secure -and $Secure.SecureSecondaries -eq 0) {
                        $VulnZonesReg += "$($Z.PSChildName)"
                    }
                }

                if ($VulnZonesReg.Count -gt 0) {
                    $script:RESULT[$CODE] = "취약"
                    $script:DETAIL[$CODE] = "Zone Transfer가 '모두 허용(0)'으로 설정된 영역이 있습니다: $($VulnZonesReg -join ', ')"
                } else {
                    $script:RESULT[$CODE] = "양호"
                    $script:DETAIL[$CODE] = "DNS 서비스 실행 중. (레지스트리 점검 결과: 영역 전송 제한됨)"
                }
            } else {
                $script:RESULT[$CODE] = "체크필요"
                $script:DETAIL[$CODE] = "DNS 서비스 실행 중이나, 설정을 확인할 수 없습니다. (권한 부족 또는 모듈 미설치)"
            }
        }
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-28. 터미널 서비스 암호화 수준 설정
# --------------------------------------------------------------------------------
function W_28 {
    $CODE = "W-28"; $script:NAME[$CODE] = "터미널 서비스 암호화 수준 설정"
    
    # 1. 원격 데스크톱(RDP) 사용 여부 확인
    # fDenyTSConnections -> 1: 차단(사용안함), 0: 허용(사용함)
    $TSPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Terminal Server"
    $TSProp = Get-ItemProperty -Path $TSPath -Name "fDenyTSConnections" -ErrorAction SilentlyContinue

    if ($TSProp -and $TSProp.fDenyTSConnections -eq 1) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "원격 데스크톱 서비스를 사용하지 않습니다."
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }

    # 2. 사용 중이라면 암호화 수준(MinEncryptionLevel) 확인
    # 경로: HKLM:\SYSTEM\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp
    # 값 설명: 1(Low), 2(Client Compatible/Medium), 3(High), 4(FIPS)
    
    $RDPPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp"
    $RDPProp = Get-ItemProperty -Path $RDPPath -Name "MinEncryptionLevel" -ErrorAction SilentlyContinue

    # 값이 없으면 0으로 간주
    $Level = if ($RDPProp -and $RDPProp.MinEncryptionLevel) { $RDPProp.MinEncryptionLevel } else { 0 }

    # [판단 기준] 2(중간) 이상이면 양호
    if ($Level -ge 2) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "암호화 수준이 '클라이언트 호환(2)' 이상으로 설정되어 있습니다. (현재값: $Level)"
    } else {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "암호화 수준이 낮거나 설정되어 있지 않습니다. (현재값: $Level, 기준: 2 이상)"
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-30. SNMP Community String 복잡성 설정
# --------------------------------------------------------------------------------
function W_30 {
    $CODE = "W-30"; $script:NAME[$CODE] = "SNMP Community String 복잡성 설정"
    
    # 1. 서비스 사용 여부 확인
    $SNMP = Get-Service "SNMP" -ErrorAction SilentlyContinue
    if ($null -eq $SNMP -or $SNMP.Status -ne "Running") {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "SNMP 서비스를 사용하지 않아 점검이 필요 없습니다."
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }

    # 2. 레지스트리에서 Community String 확인
    # 경로: HKLM:\SYSTEM\CurrentControlSet\Services\SNMP\Parameters\ValidCommunities
    # 값의 이름(Name)이 Community String입니다. (데이터는 권한)
    $RegPath = "HKLM:\SYSTEM\CurrentControlSet\Services\SNMP\Parameters\ValidCommunities"
    $Reg = Get-ItemProperty -Path $RegPath -ErrorAction SilentlyContinue

    if ($Reg) {
        # 기본값(public, private) 포함 여부 검사 (대소문자 구분 없이)
        # PSObject의 속성 이름들(Community String들)을 가져옴
        $Communities = $Reg.PSObject.Properties.Name | Where-Object { $_ -notin "PSPath", "PSParentPath", "PSChildName", "PSDrive", "PSProvider" }
        
        $WeakStrings = @()
        foreach ($Comm in $Communities) {
            if ($Comm -match "^public$" -or $Comm -match "^private$") {
                $WeakStrings += $Comm
            }
        }

        if ($WeakStrings.Count -gt 0) {
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "기본 Community String(public/private)이 설정되어 있습니다: $($WeakStrings -join ', ')"
        } else {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "SNMP 서비스 사용 중이나, 기본값(public/private)을 사용하지 않고 있습니다."
        }
    } else {
        # 레지스트리 키가 없으면 설정된 커뮤니티가 없는 것이므로(접속 불가) 양호 간주 or 확인필요
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "설정된 Community String이 없습니다."
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}



# --------------------------------------------------------------------------------
# W-31. SNMP Access Control 설정 (접근 허용 호스트 제한)
# --------------------------------------------------------------------------------
function W_31 {
    $CODE = "W-31"; $script:NAME[$CODE] = "SNMP Access Control 설정"
    
    # 1. 서비스 사용 여부 확인
    $SNMP = Get-Service "SNMP" -ErrorAction SilentlyContinue
    if ($null -eq $SNMP -or $SNMP.Status -ne "Running") {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "SNMP 서비스를 사용하지 않아 점검이 필요 없습니다."
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }

    # 2. 레지스트리에서 PermittedManagers 확인
    # 경로: HKLM:\SYSTEM\CurrentControlSet\Services\SNMP\Parameters\PermittedManagers
    # 이 키에 1, 2, 3... 등의 이름으로 허용된 IP가 등록됩니다.
    # 키가 비어있거나 없으면 "모든 호스트로부터 허용" 상태일 수 있음(취약)
    
    $RegPath = "HKLM:\SYSTEM\CurrentControlSet\Services\SNMP\Parameters\PermittedManagers"
    $Reg = Get-ItemProperty -Path $RegPath -ErrorAction SilentlyContinue

    # 실제 등록된 매니저(IP) 개수 확인
    # PS 기본 속성을 제외한 나머지 속성이 있는지 확인
    $HasManagers = $false
    if ($Reg) {
        $Props = $Reg.PSObject.Properties.Name | Where-Object { $_ -notin "PSPath", "PSParentPath", "PSChildName", "PSDrive", "PSProvider" }
        if ($Props.Count -gt 0) {
            $HasManagers = $true
        }
    }

    if ($HasManagers) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "SNMP 패킷 수신이 특정 호스트(IP)로 제한되어 있습니다."
    } else {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "SNMP 접근 제어(PermittedManagers)가 설정되지 않아 모든 호스트의 접속을 허용할 수 있습니다."
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}


# --------------------------------------------------------------------------------
# W-32. DNS 서비스 구동 점검 (동적 업데이트 제한 확인)
# --------------------------------------------------------------------------------
function W_32 {
    $CODE = "W-32"; $script:NAME[$CODE] = "DNS 서비스 구동 점검"
    
    # 1. DNS 서비스(DNS Server) 실행 여부 확인
    $DnsSvc = Get-Service "DNS" -ErrorAction SilentlyContinue

    if ($null -eq $DnsSvc -or $DnsSvc.Status -ne "Running") {
        # DNS 서비스를 안 쓰면 당연히 안전 (양호)
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "DNS 서비스를 사용하지 않습니다. (서비스 중지됨)"
    } else {
        # 2. 서비스 실행 중 -> 동적 업데이트(Dynamic Update) 설정 확인
        # 가이드 기준: "없음(None)"이어야 양호.
        
        $VulnZones = @()
        $CheckDone = $false

        # (A) PowerShell DNS 모듈 사용 (정확도 높음)
        if (Get-Command Get-DnsServerZone -ErrorAction SilentlyContinue) {
            try {
                $Zones = Get-DnsServerZone
                foreach ($Zone in $Zones) {
                    # DynamicUpdate 속성: None(0), NonSecureAndSecure(1), Secure(2)
                    # 기준에 따라 'None'이 아니면 모두 취약으로 판단
                    if ($Zone.DynamicUpdate -ne "None") {
                        $VulnZones += "$($Zone.ZoneName)($($Zone.DynamicUpdate))"
                    }
                }
                $CheckDone = $true
            } catch {
                # 모듈 오류 시 레지스트리 검사로 진행
                $CheckDone = $false
            }
        }

        # (B) 레지스트리 검사 (모듈 실패 또는 미설치 시)
        if (-not $CheckDone) {
            $RegPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\DNS Server\Zones"
            if (Test-Path $RegPath) {
                $Zones = Get-ChildItem $RegPath
                foreach ($Z in $Zones) {
                    $Prop = Get-ItemProperty -Path $Z.PSPath -Name "AllowUpdate" -ErrorAction SilentlyContinue
                    # AllowUpdate 값: 0(None), 1(NonSecure+Secure), 2(SecureOnly)
                    # 0이 아니면 취약
                    if ($Prop -and $Prop.AllowUpdate -ne 0) {
                        $TypeStr = if ($Prop.AllowUpdate -eq 2) { "보안전용" } else { "허용" }
                        $VulnZones += "$($Z.PSChildName)($TypeStr)"
                    }
                }
            }
        }

        if ($VulnZones.Count -gt 0) {
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "동적 업데이트가 활성화된 영역이 있습니다: $($VulnZones -join ', ')"
        } else {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "DNS 서비스 실행 중이나, 모든 영역의 동적 업데이트가 '없음(None)'으로 설정되어 안전합니다."
        }
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

    if ($SmtpSvc -and $SmtpSvc.Status -eq "Running") {
        # SMTP는 레거시 서비스라 스크립트로 배너 설정값(ConnectResponse) 확인이 까다로움(ADSI 필요)
        # 서비스가 켜져 있으면 기본적으로 배너가 노출될 확률이 높으므로 '체크필요' 또는 '취약'으로 분류
        $IsVuln = $true
        $VulnDetails += "SMTP:서비스실행중(배너설정수동점검필요)"
    }

    # 3. HTTP 서비스 점검 (IIS 웹 서버)
    $W3Svc = Get-Service "W3SVC" -ErrorAction SilentlyContinue
    if ($W3Svc -and $W3Svc.Status -eq "Running") {
        if (Get-Module -ListAvailable -Name WebAdministration) {
            try {
                # HTTP 응답 헤더 중 'X-Powered-By'가 존재하는지 확인
                # 경로: system.webServer/httpProtocol/customHeaders
                $Headers = Get-WebConfigurationProperty -Filter /system.webServer/httpProtocol/customHeaders -Name collection -PSPath IIS:\ -ErrorAction SilentlyContinue
                
                $HasXPoweredBy = $false
                foreach ($H in $Headers) {
                    if ($H.name -eq "X-Powered-By") {
                        $HasXPoweredBy = $true
                        break
                    }
                }

                if ($HasXPoweredBy) {
                    $IsVuln = $true
                    $VulnDetails += "HTTP:X-Powered-By헤더노출"
                }
                
                # 'Server' 헤더 차단(URL Rewrite) 여부는 스크립트 확인이 복잡하므로 
                # X-Powered-By가 없으면 일단 양호로 보되, 상세 내용에 언급
            } catch {
                $IsVuln = $true
                $VulnDetails += "HTTP:설정확인불가"
            }
        }
    }

    # 4. 결과 종합 판정
    if ($IsVuln) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "배너 차단 미흡: " + ($VulnDetails -join ", ")
    } else {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "모든 서비스(HTTP/FTP/SMTP)가 중지되었거나 배너 차단 설정이 적용되었습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-36. 원격터미널 접속 타임아웃 설정
# --------------------------------------------------------------------------------
function W_36 {
    $CODE = "W-36"; $script:NAME[$CODE] = "원격터미널 접속 타임아웃 설정"
    
    # 1. RDP 서비스 미사용 시 양호
    $TSProp = Get-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Control\Terminal Server" -Name "fDenyTSConnections" -ErrorAction SilentlyContinue
    if ($TSProp -and $TSProp.fDenyTSConnections -eq 1) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "원격 데스크톱 서비스를 사용하지 않습니다."
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }

    # 2. 타임아웃 설정 확인 (MaxIdleTime)
    # 우선순위: 그룹 정책(Policies) > 기본 RDP 설정(RDP-Tcp)
    # 단위: 밀리초(ms). 30분 = 1,800,000 ms
    
    $LimitValues = @()
    

    # (A) 정책 설정 확인
    $PolicyPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows NT\Terminal Services"
    $Policy = Get-ItemProperty -Path $PolicyPath -Name "MaxIdleTime" -ErrorAction SilentlyContinue
    if ($Policy -and $Policy.MaxIdleTime) { $LimitValues += $Policy.MaxIdleTime }

    # (B) RDP-Tcp 설정 확인
    $RdpPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp"
    $Rdp = Get-ItemProperty -Path $RdpPath -Name "MaxIdleTime" -ErrorAction SilentlyContinue
    if ($Rdp -and $Rdp.MaxIdleTime) { $LimitValues += $Rdp.MaxIdleTime }

    # 유효한 설정값이 하나라도 있는지 확인
    $IsSafe = $false
    $CurrentSetting = "설정없음"

    foreach ($Val in $LimitValues) {
        if ($Val -gt 0 -and $Val -le 1800000) { # 0초 초과 ~ 30분 이하
            $IsSafe = $true
            $CurrentSetting = "$($Val / 60000)분"
            break
        } elseif ($Val -gt 1800000) {
            $CurrentSetting = "$($Val / 60000)분(30분초과)"
        }
    }

    if ($IsSafe) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "유휴 세션 제한 시간이 30분 이하로 설정되어 있습니다. ($CurrentSetting)"
    } else {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "유휴 세션 제한(Timeout)이 설정되지 않았거나 30분을 초과합니다. (현재: $CurrentSetting)"
    }
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# --------------------------------------------------------------------------------
# W-40. 정책에 따른 시스템 로깅 설정
# --------------------------------------------------------------------------------
function W_40 {
    $CODE = "W-40"; $script:NAME[$CODE] = "정책에 따른 시스템 로깅 설정"
    
    # auditpol 명령어를 사용하여 현재 감사 정책 상태 조회
    # CSV 형태로 가져와서 분석 (성공/실패 여부 확인)
    # 주요 점검 항목: 계정 관리, 로그인 이벤트, 계정 로그인, 권한 사용, 정책 변경 등
    
    $AuditVuln = @()
    
    # auditpol이 없는 경우(드물지만) 예외 처리
    if (-not (Get-Command "auditpol.exe" -ErrorAction SilentlyContinue)) {
        $script:RESULT[$CODE] = "체크필요"
        $script:DETAIL[$CODE] = "auditpol 명령어를 찾을 수 없어 점검할 수 없습니다."
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }

    # CSV 포맷으로 감사 정책 가져오기 (/r)
    # 한글 윈도우: "감사 안 함", "성공", "실패", "성공 및 실패"
    # 영문 윈도우: "No Auditing", "Success", "Failure", "Success and Failure"
    try {
        $RawPolicy = auditpol /get /category:* /r | ConvertFrom-Csv
        
        # 점검할 주요 카테고리 (Subcategory GUID 또는 이름 매칭)
        # 이름 매칭은 언어별로 다르므로 주의 필요하나, 여기서는 "감사 안 함(No Auditing)" 상태인지만 체크하여 범용성 확보
        
        foreach ($Item in $RawPolicy) {
            # 주요 보안 카테고리 키워드 (한/영)
            $TargetCategories = "Account Management", "계정 관리", "Logon", "로그온", "Privilege Use", "권한 사용", "Policy Change", "정책 변경"
            
            # Subcategory 이름이 주요 대상에 포함되는지 확인
            $IsTarget = $false
            foreach ($Cat in $TargetCategories) {
                if ($Item.Subcategory -match $Cat) { $IsTarget = $true; break }
            }

            if ($IsTarget) {
                # 설정값(Inclusion Setting) 확인
                # "No Auditing" 또는 "감사 안 함" 이면 취약
                if ($Item."Inclusion Setting" -match "No Auditing" -or $Item."Inclusion Setting" -match "감사 안 함") {
                    $AuditVuln += "$($Item.Subcategory)"
                }
            }
        }

        if ($AuditVuln.Count -gt 0) {
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "주요 감사 정책이 설정되지 않았습니다: $($AuditVuln -join ', ')"
        } else {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "주요 이벤트(로그온, 계정관리, 정책변경 등)에 대한 감사 로깅이 활성화되어 있습니다."
        }
    } catch {
        $script:RESULT[$CODE] = "체크필요"
        $script:DETAIL[$CODE] = "감사 정책을 조회하는 중 오류가 발생했습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

        $script:DETAIL[$CODE] = "주요 이벤트 로그(App, Sec, Sys)의 최대 크기가 10MB 이상으로 설정되어 있습니다."

# --------------------------------------------------------------------------------
# W-43. 이벤트 로그 파일 접근 통제 설정
# --------------------------------------------------------------------------------
function W_43 {
    $CODE = "W-43"; $script:NAME[$CODE] = "이벤트 로그 파일 접근 통제 설정"
    
    # 점검 대상 경로:
    # 1. 최신 윈도우(2008 이상): C:\Windows\System32\winevt\Logs
    # 2. 레거시/설정 경로: C:\Windows\System32\config
    
    $TargetPaths = @(
        "$Env:SystemRoot\System32\winevt\Logs",
        "$Env:SystemRoot\System32\config"
    )
    
    $VulnPaths = @()
    
    foreach ($Path in $TargetPaths) {
        if (Test-Path $Path) {
            try {
                $Acl = Get-Acl -Path $Path -ErrorAction SilentlyContinue
                if ($Acl) {
                    # 'Everyone' 권한이 포함되어 있는지 확인 (SID: S-1-1-0)
                    $HasEveryone = $Acl.Access | Where-Object { 
                        $_.IdentityReference -match "Everyone" -or $_.IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value -eq "S-1-1-0"
                    }
                    
                    if ($HasEveryone) {
                        $VulnPaths += $Path
                    }
                }
            } catch {
                # 권한 문제로 읽지 못하는 경우 건너뜀
            }
        }
    }

    if ($VulnPaths.Count -gt 0) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "로그 디렉터리에 'Everyone' 권한이 설정되어 있습니다: $($VulnPaths -join ', ')"

# W-47. 화면 보호기 설정
# --------------------------------------------------------------------------------
function W_47 {
    $CODE = "W-47"; $script:NAME[$CODE] = "화면 보호기 설정"
    
    # 주의: 화면 보호기는 '현재 로그인한 사용자(HKCU)' 기준입니다.
    # 스크립트를 실행하는 계정의 설정이 점검되므로, 실제 운영자와 다를 수 있습니다.
    
    $Path = "HKCU:\Control Panel\Desktop"
    $Prop = Get-ItemProperty -Path $Path -ErrorAction SilentlyContinue
    
    if ($Prop) {
        $Active  = $Prop.ScreenSaveActive   # 1: 활성, 0: 비활성
        $Secure  = $Prop.ScreenSaverIsSecure # 1: 암호 사용, 0: 미사용
        $Timeout = $Prop.ScreenSaveTimeOut  # 초 단위 (600초 = 10분)

        # 값이 없는 경우 기본값 처리 (일반적으로 없으면 비활성)
        if ($null -eq $Active) { $Active = 0 }
        if ($null -eq $Secure) { $Secure = 0 }
        if ($null -eq $Timeout) { $Timeout = 99999 } 

        # [판단 기준]
        # 1. 활성화(1) 되어 있어야 함
        # 2. 암호(1) 걸려 있어야 함
        # 3. 시간(Timeout) <= 600 (10분)
        
        if ($Active -eq "1" -and $Secure -eq "1" -and $Timeout -le 600) {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "화면 보호기가 보안 기준(활성, 암호사용, 10분 이하)을 만족합니다."

            if ($Active -ne "1") { $Reasons += "미사용" }
            if ($Secure -ne "1") { $Reasons += "암호미설정" }
            if ($Timeout -gt 600) { $Reasons += "시간초과($([math]::Round($Timeout/60))분)" }
            
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "화면 보호기 설정 미흡: " + ($Reasons -join ", ")
        }
    } else {
        $script:RESULT[$CODE] = "체크필요"
        $script:DETAIL[$CODE] = "화면 보호기 설정을 읽을 수 없습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-54. DoS 공격 방어 레지스트리 설정
# --------------------------------------------------------------------------------
function W_54 {
    $CODE = "W-54"; $script:NAME[$CODE] = "DoS 공격 방어 레지스트리 설정"
    
    # 레지스트리: HKLM:\SYSTEM\CurrentControlSet\Services\Tcpip\Parameters
    # 점검 항목 4가지 (이미지 기준)
    # 1. SynAttackProtect >= 1 (기본값 0 또는 없음 -> 취약)
    # 2. EnableDeadGWDetect = 0 (기본값 1 -> 취약)
    # 3. KeepAliveTime = 300,000 (5분) (기본값 7,200,000(2시간) -> 취약)
    # 4. NoNameReleaseOnDemand = 1 (기본값 0 -> 취약)

    $Path = "HKLM:\SYSTEM\CurrentControlSet\Services\Tcpip\Parameters"
    $Reg = Get-ItemProperty -Path $Path -ErrorAction SilentlyContinue
    
    if (-not $Reg) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "TCP/IP 파라미터 레지스트리 경로를 찾을 수 없습니다."
        print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
        return
    }

    $VulnParams = @()

    # 1. SynAttackProtect (없으면 0으로 간주)
    $Syn = if ($Reg.SynAttackProtect) { $Reg.SynAttackProtect } else { 0 }
    if ($Syn -lt 1) { $VulnParams += "SynAttackProtect($Syn)" }

    # 2. EnableDeadGWDetect (없으면 1로 간주 - 윈도우 기본값)
    $DeadGW = if ($Reg.EnableDeadGWDetect) { $Reg.EnableDeadGWDetect } else { 1 }
    if ($DeadGW -ne 0) { $VulnParams += "EnableDeadGWDetect($DeadGW)" }

    # 3. KeepAliveTime (없으면 7,200,000으로 간주)
    $KeepAlive = if ($Reg.KeepAliveTime) { $Reg.KeepAliveTime } else { 7200000 }
    # 가이드라인은 300,000(5분) 권장. 300,000 보다 크면 취약으로 판단
    if ($KeepAlive -gt 300000) { $VulnParams += "KeepAliveTime($([math]::Round($KeepAlive/60000))분)" }

    # 4. NoNameReleaseOnDemand (없으면 0으로 간주)
    $NoName = if ($Reg.NoNameReleaseOnDemand) { $Reg.NoNameReleaseOnDemand } else { 0 }
    if ($NoName -ne 1) { $VulnParams += "NoNameReleaseOnDemand($NoName)" }

    if ($VulnParams.Count -gt 0) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "DoS 방어 설정 미흡: " + ($VulnParams -join ", ")
    } else {
        $script:RESULT[$CODE] = "양호"

# --------------------------------------------------------------------------------
# W-56. SMB 세션 중단 관리 설정
# --------------------------------------------------------------------------------
function W_56 {
    $CODE = "W-56"; $script:NAME[$CODE] = "SMB 세션 중단 관리 설정"
    
    # 경로: HKLM:\SYSTEM\CurrentControlSet\Services\LanmanServer\Parameters
    # 점검 항목 2가지 (AND 조건)
    # 1. enableforcedlogoff = 1 (로그온 시간 만료 시 연결 끊기: 사용)
    # 2. autodisconnect <= 15 (유휴 시간: 15분 이하)
    
    $Path = "HKLM:\SYSTEM\CurrentControlSet\Services\LanmanServer\Parameters"
    $Reg = Get-ItemProperty -Path $Path -ErrorAction SilentlyContinue
    
    if ($Reg) {
        $Forced = if ($Reg.enableforcedlogoff) { $Reg.enableforcedlogoff } else { 0 }
        $AutoDisc = if ($Reg.autodisconnect) { $Reg.autodisconnect } else { 15 } # 없으면 기본값(15) 가정
        
        $IsVuln = $false
        $Details = @()

        # 1. 연결 끊기 설정 확인
        if ($Forced -ne 1) {
            $IsVuln = $true
            $Details += "만료시연결끊기미사용"
        }

        # 2. 유휴 시간 확인 (0은 끊지 않음이므로 취약, 15분 초과 시 취약)
        if ($AutoDisc -eq 0 -or $AutoDisc -gt 15) {
            $IsVuln = $true
            $Details += "유휴시간($($AutoDisc)분)"
        }

        if ($IsVuln) {
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "SMB 세션 설정 미흡: " + ($Details -join ", ")
        } else {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "SMB 세션 만료 시 연결 차단 및 유휴 시간($($AutoDisc)분)이 적절히 설정되어 있습니다."
        }
    } else {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "LanmanServer 파라미터 레지스트리를 확인할 수 없습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# --------------------------------------------------------------------------------
# W-58. 사용자별 홈 디렉터리 권한 설정
# --------------------------------------------------------------------------------
function W_58 {
    $CODE = "W-58"; $script:NAME[$CODE] = "사용자별 홈 디렉터리 권한 설정"
    
    # 점검 대상: C:\Users 하위 폴더들
    # 제외 대상: Public, Default, All Users, Default User 등 공용 폴더
    # 점검 내용: ACL에 'Everyone' 권한이 있는지 확인
    
    $BaseDir = "C:\Users"
    $ExcludeList = @("Public", "Default", "All Users", "Default User", "Administrator") # Admin은 W-58 점검대상에서 보통 제외하거나 별도 관리
    
    $VulnDirs = @()
    
    if (Test-Path $BaseDir) {
        $UserDirs = Get-ChildItem -Path $BaseDir -Directory | Where-Object { $_.Name -notin $ExcludeList }
        
        foreach ($Dir in $UserDirs) {
            try {
                $Acl = Get-Acl -Path $Dir.FullName -ErrorAction SilentlyContinue
                
                # Everyone (S-1-1-0) 권한 확인
                $HasEveryone = $false
                foreach ($Access in $Acl.Access) {
                    if ($Access.IdentityReference -match "Everyone" -or $Access.IdentityReference.Value -eq "S-1-1-0") {
                        $HasEveryone = $true
                        break
                    }
                }
                
                if ($HasEveryone) {
                    $VulnDirs += $Dir.Name
                }
            } catch {
                # 권한 부족 등으로 읽지 못한 경우 패스
            }
        }

        if ($VulnDirs.Count -gt 0) {
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "홈 디렉터리에 Everyone 권한이 포함된 계정이 있습니다: $($VulnDirs -join ', ')"
        } else {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "사용자 홈 디렉터리에 Everyone 권한이 발견되지 않았습니다."
        }
    } else {
        $script:RESULT[$CODE] = "체크필요"
        $script:DETAIL[$CODE] = "C:\Users 디렉터리를 찾을 수 없습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-60. 보안 채널 데이터 디지털 암호화 또는 서명
# --------------------------------------------------------------------------------
function W_60 {
    $CODE = "W-60"; $script:NAME[$CODE] = "보안 채널 데이터 디지털 암호화 또는 서명"
    
    # 레지스트리 경로: HKLM:\SYSTEM\CurrentControlSet\Services\Netlogon\Parameters
    # 점검해야 할 3가지 항목 (모두 1이어야 양호)
    # 1. RequireSignOrSeal: 암호화 또는 서명 (항상)
    # 2. SealSecureChannel: 암호화 (가능한 경우)
    # 3. SignSecureChannel: 서명 (가능한 경우)
    
    $Path = "HKLM:\SYSTEM\CurrentControlSet\Services\Netlogon\Parameters"
    $Reg = Get-ItemProperty -Path $Path -ErrorAction SilentlyContinue
    
    if ($Reg) {
        $Require = if ($Reg.RequireSignOrSeal) { $Reg.RequireSignOrSeal } else { 1 } # 기본값 1(최신 OS)
        $Seal    = if ($Reg.SealSecureChannel) { $Reg.SealSecureChannel } else { 1 }
        $Sign    = if ($Reg.SignSecureChannel) { $Reg.SignSecureChannel } else { 1 }
        
        if ($Require -eq 1 -and $Seal -eq 1 -and $Sign -eq 1) {
            $script:RESULT[$CODE] = "양호"
            $script:DETAIL[$CODE] = "보안 채널 데이터의 암호화 및 서명 정책이 모두 활성화되어 있습니다."
        } else {
            $Missing = @()
            if ($Require -ne 1) { $Missing += "항상 암호화/서명 미설정" }
            if ($Seal -ne 1) { $Missing += "암호화(가능시) 미설정" }
            if ($Sign -ne 1) { $Missing += "서명(가능시) 미설정" }
            
            $script:RESULT[$CODE] = "취약"
            $script:DETAIL[$CODE] = "보안 채널 설정 미흡: " + ($Missing -join ", ")
        }
    } else {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "Netlogon 파라미터 레지스트리를 확인할 수 없습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

# W-62. 시작 프로그램 목록 분석
# --------------------------------------------------------------------------------
function W_62 {
    $CODE = "W-62"; $script:NAME[$CODE] = "시작 프로그램 목록 분석"
    
    # 시작 프로그램 목록을 추출하여 보여주고, 관리자가 판단하도록 '체크필요' 처리
    # 점검 위치: 레지스트리 Run/RunOnce (HKLM, HKCU)
    
    $StartupItems = @()
    
    $RegPaths = @(
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Run",
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\RunOnce",
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run",
        "HKCU:\Software\Microsoft\Windows\CurrentVersion\RunOnce"
    )
    
    foreach ($P in $RegPaths) {
        $Items = Get-ItemProperty -Path $P -ErrorAction SilentlyContinue
        if ($Items) {
            # 기본 속성 제외하고 실제 등록된 값만 추출
            $Props = $Items.PSObject.Properties.Name | Where-Object { $_ -notin "PSPath", "PSParentPath", "PSChildName", "PSDrive", "PSProvider" }
            foreach ($Key in $Props) {
                $StartupItems += "$Key"
            }
        }
    }
    
    # 시작 폴더 항목도 추가하려면 Get-ChildItem "$env:APPDATA\Microsoft\Windows\Start Menu\Programs\Startup" 등을 쓸 수 있음
    
    $Count = $StartupItems.Count
    
    $script:RESULT[$CODE] = "체크필요"
    if ($Count -gt 0) {
        # 항목이 너무 많으면 일부만 표시
        $DisplayItems = if ($Count -gt 5) { ($StartupItems[0..4] -join ", ") + " 등 총 $Count 개" } else { $StartupItems -join ", " }
        $script:DETAIL[$CODE] = "[수동 점검 필요] 등록된 시작 프로그램: $DisplayItems. 불필요한 항목이 있는지 확인하세요."
    } else {

    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}

function W_63 {
    $CODE = "W-63"; $script:NAME[$CODE] = "도메인 컨트롤러-사용자의 시간 동기화"
    
    # 이 설정은 'Kerberos 정책'으로, 도메인 컨트롤러(DC)이거나 도메인에 가입된 서버에서 중요합니다.
    # secedit 명령어를 통해 보안 정책을 내보내서 확인합니다.
    
    $ExportFile = "$env:TEMP\secpol_w63.cfg"
    secedit /export /cfg $ExportFile /areas SECURITYPOLICY | Out-Null
    
    $MaxClockSkew = 5 # 기본값 (분)

                $MaxClockSkew = [int]$matches[1]
                $Found = $true
                break
            }
        }
        Remove-Item $ExportFile -ErrorAction SilentlyContinue
    }

    # 기준: 5분 이하 (양호)
    if ($MaxClockSkew -le 5) {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "Kerberos 컴퓨터 시계 동기화 최대 허용 오차가 $($MaxClockSkew)분으로 설정되어 있습니다."
    } else {

        $script:DETAIL[$CODE] = "시계 동기화 허용 오차가 5분을 초과합니다. (현재: $($MaxClockSkew)분)"

# W-64. 윈도우 방화벽 설정
# --------------------------------------------------------------------------------
function W_64 {
    $CODE = "W-64"; $script:NAME[$CODE] = "윈도우 방화벽 설정"
    
    # Windows 방화벽(Domain, Private, Public) 활성화 여부 확인
    # PowerShell 모듈(NetSecurity) 사용 (Server 2012 이상)
    # 구형 서버 호환성을 위해 netsh 명령어도 고려 가능하나, 최신 환경 기준 모듈 우선 사용
    
    $VulnProfiles = @()
    
    if (Get-Command Get-NetFirewallProfile -ErrorAction SilentlyContinue) {
        $Profiles = Get-NetFirewallProfile
        foreach ($P in $Profiles) {
            if ($P.Enabled -ne $true -and $P.Enabled -ne "True") {
                $VulnProfiles += "$($P.Name)(Off)"
            }
        }
    } else {
        # 모듈이 없는 구형 OS의 경우 netsh 사용
        $NetshResult = netsh advfirewall show allprofiles state
        # 결과 문자열 파싱이 필요하지만, 여기서는 단순화하여 'OFF'가 포함되면 취약 의심으로 처리하거나
        # "State ON"이 3개 미만이면 취약으로 간주
        
        # 간단히 보수적으로 '체크필요'로 넘기지 않고, ON이 하나라도 없으면 잡도록 로직 구성
        $OnCount = ($NetshResult | Select-String "ON").Count
        if ($OnCount -lt 3) { # 프로필 3개(Domain, Private, Public) 모두 ON이어야 함
             $VulnProfiles += "일부 프로필 Off 확인됨"
        }
    }

    if ($VulnProfiles.Count -gt 0) {
        $script:RESULT[$CODE] = "취약"
        $script:DETAIL[$CODE] = "Windows 방화벽이 비활성화된 프로필이 있습니다: $($VulnProfiles -join ', ')"
    } else {
        $script:RESULT[$CODE] = "양호"
        $script:DETAIL[$CODE] = "모든 네트워크 프로필(Domain, Private, Public)에 대해 방화벽이 활성화되어 있습니다."
    }
    
    print_result $CODE $script:NAME[$CODE] $script:RESULT[$CODE] $script:DETAIL[$CODE]
}
